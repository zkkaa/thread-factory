function isRunnable(thread: ThreadRuntime): boolean {
  return thread.state === 'READY' || thread.state === 'RUNNING';
}

/**
 * Mesin simulasi. Tidak punya timer: hanya maju saat step() dipanggil
 * (oleh SimulationController). Semua keacakan berasal dari PRNG berbasis seed,
 * jadi seed yang sama menghasilkan jalannya simulasi yang sama persis.
 *
 * Satu tick:
 *  1. thread WAITING menambah waitTicks; thread NEW dimasukkan ke READY
 *  2. scheduler memilih hingga `cores` thread yang boleh jalan
 *  3. thread RUNNING yang tidak terpilih turun ke READY (context switch)
 *  4. thread terpilih menjadi RUNNING dan mengeksekusi SATU instruksi
 *  5. simulasi selesai bila semua TERMINATED atau tick mencapai maxTicks
 */
export class Engine implements SimulationEngine {
  private config: EngineConfig | null = null;
  private seed = 0;
  private tick = 0;
  private finished = false;
  private executedInstructions = 0;
  private threads: ThreadRuntime[] = [];
  private threadById = new Map<string, ThreadRuntime>();
  private mutexes = new Map<string, Mutex>();
  private semaphores = new Map<string, Semaphore>();
  private conditions = new Map<string, Condition>();
  private buffers = new Map<string, BoundedBuffer>();
  private rwLocks = new Map<string, RwLock>();
  private boards = new Map<string, Board>();
  private sync: SyncContext | null = null;
  private sharedVars: Record<string, number> = {};
  private scheduler: Scheduler | null = null;
  private areaMembers = new Map<string, Set<string>>();
  private areaCapacities = new Map<string, number>();
  private overloadTickCount = 0;
  private areaOccupancySum = new Map<string, number>();
  // Starvation (T-053): rentetan menunggu saat ini dan puncaknya per thread.
  private waitStreaks = new Map<string, number>();
  private peakWaitStreaks = new Map<string, number>();

  get isFinished(): boolean {
    return this.finished;
  }

  load(config: EngineConfig, seed: number): void {
    // Semua pemeriksaan dilakukan dulu agar load() yang gagal tidak merusak state lama.
    validateConfig(config);
    const prng = createPrng(seed);
    const scheduler = createScheduler(config.scheduler, prng);

    this.config = config;
    this.seed = seed;
    this.tick = 0;
    this.finished = false;
    this.executedInstructions = 0;
    this.sharedVars = { ...config.sharedVars };
    this.threads = config.threads.map(createThread);

    const threadById = new Map(this.threads.map((t) => [t.def.id, t]));
    this.threadById = threadById;
    this.waitStreaks = new Map(this.threads.map((t) => [t.def.id, 0]));
    this.peakWaitStreaks = new Map(this.threads.map((t) => [t.def.id, 0]));

    this.mutexes = new Map(
      config.mutexes.map((name) => [name, new Mutex(name)]),
    );

    this.semaphores = new Map(
      Object.entries(config.semaphores).map(([name, val]) => [
        name,
        new Semaphore(name, val),
      ]),
    );

    this.conditions = new Map(
      config.conditions.map((name) => [name, new Condition(name)]),
    );

    this.buffers = new Map(
      Object.entries(config.buffers ?? {}).map(([name, capacity]) => [
        name,
        new BoundedBuffer(name, capacity),
      ]),
    );

    this.rwLocks = new Map(
      Object.entries(config.rwLocks ?? {}).map(([name, policy]) => [
        name,
        new RwLock(name, policy),
      ]),
    );

    this.boards = new Map(
      (config.boards ?? []).map((name) => [name, new Board(name)]),
    );

    this.areaCapacities = new Map(Object.entries(config.areas ?? {}));

    this.areaMembers = new Map(
      Array.from(this.areaCapacities.keys(), (area) => [
        area,
        new Set<string>(),
      ]),
    );

    this.overloadTickCount = 0;

    this.areaOccupancySum = new Map(
      Array.from(this.areaCapacities.keys(), (area) => [area, 0]),
    );

    this.sync = {
      mutexes: this.mutexes,
      semaphores: this.semaphores,
      conditions: this.conditions,
      buffers: this.buffers,
      rwLocks: this.rwLocks,
      boards: this.boards,

      enterArea: (threadId, area) => {
        let members = this.areaMembers.get(area);

        if (!members) {
          members = new Set();
          this.areaMembers.set(area, members);
        }

        members.add(threadId);
      },

      leaveArea: (threadId, area) => {
        this.areaMembers.get(area)?.delete(threadId);
      },

      getThread(id) {
        const thread = threadById.get(id);

        if (thread === undefined) {
          throw new Error(`Thread tidak ada: "${id}"`);
        }

        return thread;
      },
    };

    this.scheduler = scheduler;
  }

  reset(seed?: number): void {
    if (this.config === null) {
      throw new Error('reset() dipanggil sebelum load()');
    }

    this.load(this.config, seed ?? this.seed);
  }

  /**
   * Maju tepat satu tick. Bila simulasi sudah selesai, tidak terjadi apa-apa
   * (tick tidak bertambah dan events kosong).
   */
  step(): StepResult {
    if (this.config === null || this.scheduler === null || this.sync === null) {
      throw new Error('step() dipanggil sebelum load()');
    }

    if (this.finished) {
      return { tick: this.tick, events: [], finished: true };
    }

    this.tick += 1;
    const tick = this.tick;
    const events: SimEvent[] = [];

    const push = (event: SimEvent | null): void => {
      if (event) events.push(event);
    };

    // 1. Waktu tunggu dan admisi thread baru
    for (const thread of this.threads) {
      const id = thread.def.id;
      if (thread.state === 'WAITING') {
        thread.waitTicks += 1;
        const streak = (this.waitStreaks.get(id) ?? 0) + 1;
        this.waitStreaks.set(id, streak);
        if (streak > (this.peakWaitStreaks.get(id) ?? 0)) {
          this.peakWaitStreaks.set(id, streak);
        }
      } else {
        // Dibangunkan (handoff) atau tidak menunggu: rentetan putus.
        this.waitStreaks.set(id, 0);
      }
    }

    for (const thread of this.threads) {
      if (thread.state === 'NEW') {
        push(transition(thread, 'READY', tick));
      }
    }

    // 2. Pemilihan oleh scheduler
    const runnable = this.threads.filter(isRunnable).map((t) => t.def.id);

    if (runnable.length === 0) {
      this.finished = true;
      events.push({ type: 'FINISHED', tick });

      return { tick, events, finished: true };
    }

    const picked = this.scheduler.pick(runnable);
    const pickedSet = new Set(picked);

    // 3. Thread yang tidak terpilih dibekukan
    for (const thread of this.threads) {
      if (thread.state === 'RUNNING' && !pickedSet.has(thread.def.id)) {
        push(transition(thread, 'READY', tick));
      }
    }

    // 4. Thread terpilih mengeksekusi satu instruksi
    for (const id of picked) {
      const thread = this.threadById.get(id);

      if (thread === undefined) {
        throw new Error(`Scheduler memilih thread yang tidak ada: "${id}"`);
      }

      push(transition(thread, 'RUNNING', tick));
      this.executedInstructions += 1;

      events.push(
        ...executeInstruction(thread, tick, this.sharedVars, this.sync),
      );

      this.assertNoLockLeak(thread);
    }

    // 4b. Pemeriksaan kapasitas area
    let hasOverloadThisTick = false;

    for (const [area, capacity] of this.areaCapacities.entries()) {
      const count = this.areaMembers.get(area)?.size ?? 0;

      this.areaOccupancySum.set(
        area,
        (this.areaOccupancySum.get(area) ?? 0) + count,
      );

      if (count > capacity) {
        hasOverloadThisTick = true;

        events.push({
          type: 'AREA_OVERLOAD',
          tick,
          area,
          count,
          capacity,
        });
      }
    }

    if (hasOverloadThisTick) {
      this.overloadTickCount += 1;
    }

    // 5. Selesai?
    const allTerminated = this.threads.every(
      (thread) => thread.state === 'TERMINATED',
    );

    if (allTerminated || tick >= this.config.maxTicks) {
      this.finished = true;
      events.push({ type: 'FINISHED', tick });
    }

    return { tick, events, finished: this.finished };
  }

  /**
   * Thread yang selesai tidak boleh masih memegang mutex.
   */
  private assertNoLockLeak(thread: ThreadRuntime): void {
    if (thread.state !== 'TERMINATED') return;

    for (const mutex of this.mutexes.values()) {
      if (mutex.isOwnedBy(thread.def.id)) {
        throw new Error(
          `Thread "${thread.def.id}" selesai tetapi masih memegang mutex "${mutex.id}" ` +
            `(LOCK dan UNLOCK tidak seimbang)`,
        );
      }
    }
  }

  /** Salinan keadaan penuh; aman dari modifikasi pemanggil. */
  getSnapshot(): Readonly<Snapshot> {
    if (this.config === null) {
      throw new Error('getSnapshot() dipanggil sebelum load()');
    }

    let areaUtilization = 0;

    if (this.areaCapacities.size > 0 && this.tick > 0) {
      let totalRatio = 0;

      for (const [area, capacity] of this.areaCapacities.entries()) {
        const avgCount = (this.areaOccupancySum.get(area) ?? 0) / this.tick;

        const ratio = capacity > 0 ? Math.min(1, avgCount / capacity) : 0;

        totalRatio += ratio;
      }

      areaUtilization = totalRatio / this.areaCapacities.size;
    }

    const boards = Array.from(this.boards.values(), (board) =>
      board.snapshot(),
    );

    const buffers = Array.from(this.buffers.values(), (buffer) =>
      buffer.snapshot(),
    );

    return {
      tick: this.tick,
      seed: this.seed,
      finished: this.finished,
      threads: this.threads.map(snapshotThread),
      sharedVars: { ...this.sharedVars },
      mutexes: Array.from(this.mutexes.values(), (m) => m.snapshot()),
      semaphores: Array.from(this.semaphores.values(), (s) => s.snapshot()),
      conditions: Array.from(this.conditions.values(), (c) => c.snapshot()),
      areaOccupancy: Object.fromEntries(
        Array.from(this.areaMembers.entries(), ([key, value]) => [
          key,
          value.size,
        ]),
      ),
      overloadTicks: this.overloadTickCount,
      areaUtilization,
      cores: this.config.scheduler.cores,
      executedInstructions: this.executedInstructions,
      buffers,
      overflowCount: buffers.reduce((sum, buffer) => sum + buffer.overflows, 0),
      underflowCount: buffers.reduce(
        (sum, buffer) => sum + buffer.underflows,
        0,
      ),
      boards,
      rwLocks: Array.from(this.rwLocks.values(), (lock) => lock.snapshot()),
      inconsistentReadCount: boards.reduce(
        (sum, board) => sum + board.inconsistentReads,
        0,
      ),
      writeConflictCount: boards.reduce(
        (sum, board) => sum + board.writeConflicts,
        0,
      ),
      busyWaitTicks: this.threads.reduce(
        (sum, thread) => sum + thread.busyWaitTicks,
        0,
      ),
      waitStreaks: Object.fromEntries(this.waitStreaks),
      peakWaitStreaks: Object.fromEntries(this.peakWaitStreaks),
      starvationTicks: this.config.starvationTicks ?? null,
    };
  }
}

} from './types';

/**
 * Keadaan sebuah thread selama simulasi berjalan.
 * Objek ini dimiliki dan diubah HANYA oleh Engine; pihak luar hanya melihat
 * salinannya lewat snapshotThread().
 */
export interface ThreadRuntime {
  readonly def: ThreadDef;
  state: ThreadState;
  /** Indeks instruksi berikutnya. Saat WAITING, menunjuk instruksi yang memblokir. */
  pc: number;
  /** Pengulangan program ke berapa (mulai 0). */
  iteration: number;
  registers: Record<string, number>;
  waitingOn: string | null;
  /** Sisa tick untuk instruksi WORK yang sedang berjalan (0 = tidak sedang WORK). */
  workRemaining: number;
  waitTicks: number;
  busyWaitTicks: number;
}

/**
 * Akses ke resource sinkronisasi dan thread lain, dibutuhkan instruksi yang
 * bisa memblokir atau membangunkan thread (LOCK, UNLOCK, dst.).
 */
export interface SyncContext {
  mutexes: ReadonlyMap<string, Mutex>;
  semaphores?: ReadonlyMap<string, Semaphore>;
  conditions?: ReadonlyMap<string, Condition>;
  buffers?: ReadonlyMap<string, BoundedBuffer>;
  rwLocks?: ReadonlyMap<string, RwLock>;
  boards?: ReadonlyMap<string, Board>;
  enterArea?: (threadId: ThreadId, area: string) => void;
  leaveArea?: (threadId: ThreadId, area: string) => void;
  /** Mengambil thread lain (untuk dibangunkan). Melempar error bila tidak ada. */
  getThread(id: ThreadId): ThreadRuntime;
}

const EMPTY_SYNC: SyncContext = {
  mutexes: new Map(),
  getThread(id) {
    throw new Error(
      `Thread "${id}" tidak tersedia (tanpa konteks sinkronisasi)`,
    );
  },
};

/**
 * Transisi state yang sah. Selain tabel ini, engine melempar error supaya bug
 * (mis. membangunkan thread yang tidak sedang WAITING) langsung ketahuan.
 */
const ALLOWED_TRANSITIONS: Record<ThreadState, readonly ThreadState[]> = {
  NEW: ['READY'],
  READY: ['RUNNING'],
  RUNNING: ['READY', 'WAITING', 'TERMINATED'],
  WAITING: ['READY'],
  TERMINATED: [],
};

export function createThread(def: ThreadDef): ThreadRuntime {
  return {
    def,
    state: 'NEW',
    pc: 0,
    iteration: 0,
    registers: {},
    waitingOn: null,
    workRemaining: 0,
    waitTicks: 0,
    busyWaitTicks: 0,
  };
}

/** Salinan keadaan thread untuk Snapshot (aman dari modifikasi pihak luar). */
export function snapshotThread(thread: ThreadRuntime): ThreadSnapshot {
  return {
    id: thread.def.id,
    label: thread.def.label,
    state: thread.state,
    pc: thread.pc,
    iteration: thread.iteration,
    registers: { ...thread.registers },
    waitingOn: thread.waitingOn,
    workRemaining: thread.workRemaining,
    waitTicks: thread.waitTicks,
    busyWaitTicks: thread.busyWaitTicks,
  };
}

/**
 * Mengubah state thread. Mengembalikan event THREAD_STATE, atau null bila state
 * tidak berubah (mis. RUNNING -> RUNNING). Melempar error untuk transisi tidak sah.
 */
export function transition(
  thread: ThreadRuntime,
  to: ThreadState,
  tick: number,
): SimEvent | null {
  const from = thread.state;

  if (from === to) return null;

  if (!ALLOWED_TRANSITIONS[from].includes(to)) {
    throw new Error(
      `Transisi state tidak sah pada thread "${thread.def.id}": ${from} -> ${to}`,
    );
  }

  thread.state = to;

  // Keluar dari WAITING berarti tidak lagi menunggu resource apa pun.
  if (from === 'WAITING') thread.waitingOn = null;

  return {
    type: 'THREAD_STATE',
    tick,
    thread: thread.def.id,
    from,
    to,
  };
}

function hasKey(record: Record<string, number>, key: string): boolean {
  return Object.prototype.hasOwnProperty.call(record, key);
}

function readNumber(
  record: Record<string, number>,
  key: string,
  threadId: string,
  kind: string,
): number {
  if (!hasKey(record, key)) {
    throw new Error(
      `Thread "${threadId}" memakai ${kind} "${key}" yang tidak ada`,
    );
  }

  return record[key] as number;
}

function requireMutex(
  sync: SyncContext,
  name: string,
  threadId: string,
): Mutex {
  const mutex = sync.mutexes.get(name);

  if (mutex === undefined) {
    throw new Error(
      `Thread "${threadId}" memakai mutex "${name}" yang tidak ada`,
    );
  }

  return mutex;
}

function requireSemaphore(
  sync: SyncContext,
  name: string,
  threadId: string,
): Semaphore {
  const sem = sync.semaphores?.get(name);

  if (sem === undefined) {
    throw new Error(
      `Thread "${threadId}" memakai semaphore "${name}" yang tidak ada`,
    );
  }

  return sem;
}

function requireCondition(
  sync: SyncContext,
  name: string,
  threadId: string,
): Condition {
  const cond = sync.conditions?.get(name);

  if (cond === undefined) {
    throw new Error(
      `Thread "${threadId}" memakai condition variable "${name}" yang tidak ada`,
    );
  }

  return cond;
}

function requireBuffer(
  sync: SyncContext,
  name: string,
  threadId: ThreadId,
): BoundedBuffer {
  const buffer = sync.buffers?.get(name);

  if (buffer === undefined) {
    throw new Error(`Thread "${threadId}": ban "${name}" tidak dideklarasikan`);
  }

  return buffer;
}

/**
 * Memajukan pc setelah satu instruksi selesai. Bila program habis, pengulangan
 * berikutnya dimulai; bila semua pengulangan habis, thread TERMINATED
 * (thread harus RUNNING saat itu).
 */
function advancePc(thread: ThreadRuntime, tick: number): SimEvent | null {
  thread.pc += 1;

  if (thread.pc < thread.def.program.length) return null;

  thread.pc = 0;
  thread.iteration += 1;

  if (thread.iteration >= (thread.def.repeat ?? 1)) {
    return transition(thread, 'TERMINATED', tick);
  }

  return null;
}

function requireRwLock(
  sync: SyncContext,
  name: string,
  threadId: ThreadId,
): RwLock {
  const lock = sync.rwLocks?.get(name);

  if (lock === undefined) {
    throw new Error(
      `Thread "${threadId}" memakai kunci baca-tulis "${name}" yang tidak ada`,
    );
  }

  return lock;
}

function requireBoard(
  sync: SyncContext,
  name: string,
  threadId: ThreadId,
): Board {
  const board = sync.boards?.get(name);

  if (board === undefined) {
    throw new Error(
      `Thread "${threadId}": papan "${name}" tidak dideklarasikan`,
    );
  }

  return board;
}

/**
 * Thread yang dibangunkan RwLock sudah menjadi pemegang kunci (handoff, seperti
 * mutex): ia READY, instruksi acquire-nya dianggap selesai, dan pc dimajukan.
 */
function wakeRwGranted(
  lock: RwLock,
  lockName: string,
  granted: readonly ThreadId[],
  tick: number,
  sync: SyncContext,
): SimEvent[] {
  const events: SimEvent[] = [];
  const holders = lock.snapshot();

  for (const id of granted) {
    const woken = sync.getThread(id);
    const change = transition(woken, 'READY', tick);

    if (change) events.push(change);

    events.push({
      type: 'RW_ACQUIRED',
      tick,
      thread: id,
      lock: lockName,
      mode: holders.writer === id ? 'write' : 'read',
    });

    const end = advancePc(woken, tick);

    if (end) events.push(end);
  }

  return events;
}

/**
 * Pemilik melepas mutex. Dipakai UNLOCK dan COND_WAIT (yang melepas mutex
 * sebelum tidur). Bila ada thread yang antre, kepemilikan langsung diserahkan
 * (handoff): thread itu dibangunkan dan instruksinya yang memblokir (LOCK atau
 * COND_WAIT) dianggap selesai, jadi pc-nya dimajukan di sini.
 * Melempar error bila `thread` bukan pemilik.
 */
function releaseMutex(
  mutex: Mutex,
  mutexName: string,
  thread: ThreadRuntime,
  tick: number,
  sync: SyncContext,
): SimEvent[] {
  const id = thread.def.id;
  const events: SimEvent[] = [];

  const push = (event: SimEvent | null): void => {
    if (event) events.push(event);
  };

  const next = mutex.unlock(id); // melempar error bila bukan pemilik

  events.push({
    type: 'LOCK_RELEASED',
    tick,
    thread: id,
    mutex: mutexName,
  });

  if (next !== null) {
    const woken = sync.getThread(next);

    push(transition(woken, 'READY', tick));

    events.push({
      type: 'LOCK_ACQUIRED',
      tick,
      thread: next,
      mutex: mutexName,
    });

    push(advancePc(woken, tick));
  }

  return events;
}

/**
 * Membangunkan satu penunggu condition. Penunggu harus mengambil mutex-nya
 * lagi sebelum boleh lanjut:
 * - mutex bebas  -> langsung memegangnya, READY, COND_WAIT selesai;
 * - mutex dipakai -> mengantre di mutex, tetap WAITING (kini menunggu mutex).
 *   COND_WAIT-nya selesai nanti saat mutex diserahkan lewat releaseMutex().
 */
function wakeWaiter(
  waiter: ConditionWaiter,
  tick: number,
  sync: SyncContext,
): SimEvent[] {
  const events: SimEvent[] = [];
  const woken = sync.getThread(waiter.thread);
  const mutex = requireMutex(sync, waiter.mutex, waiter.thread);

  if (mutex.lock(waiter.thread)) {
    const change = transition(woken, 'READY', tick);

    if (change) events.push(change);

    events.push({
      type: 'LOCK_ACQUIRED',
      tick,
      thread: waiter.thread,
      mutex: waiter.mutex,
    });

    const end = advancePc(woken, tick);

    if (end) events.push(end);
  } else {
    const on = `mutex:${waiter.mutex}`;
    woken.waitingOn = on;

    events.push({
      type: 'BLOCKED',
      tick,
      thread: waiter.thread,
      on,
    });
  }

  return events;
}

/**
 * Menjalankan SATU instruksi pada pc saat ini (satu tick), lalu memajukan pc.
 * Thread harus sudah berstatus RUNNING. Mengubah thread, sharedVars, dan
 * (untuk instruksi sinkronisasi) mutex, semaphore, condition, serta thread
 * lain yang dibangunkan.
 *
 * Catatan perilaku:
 * - WORK n membutuhkan n tick; tiap tick tetap menghasilkan event INSTRUCTION.
 * - LOCK pada mutex terpakai membuat thread WAITING dengan pc tetap di LOCK.
 *   Instruksi itu dianggap selesai saat UNLOCK menyerahkan mutex kepadanya
 *   (handoff); saat itu pc-nya dimajukan oleh thread yang melepas.
 * - COND_WAIT melepas mutex lalu tidur di condition (WAITING, pc tetap di
 *   COND_WAIT). Setelah disinyal, ia harus mengambil mutex lagi; COND_WAIT
 *   selesai saat mutex didapat (lihat wakeWaiter). Sinyal tanpa penunggu hilang.
 * - SPIN_UNTIL mengecek variabel bersama tiap tick. Selama nilainya masih di
 *   bawah batas, pc tidak maju dan thread tidak tidur (busy waiting), sehingga
 *   busyWaitTicks bertambah dan core terpakai sia-sia.
 * - Setelah instruksi terakhir selesai, program diulang sampai `repeat`
 *   terpenuhi; setelah itu thread TERMINATED.
 */
export function executeInstruction(
  thread: ThreadRuntime,
  tick: number,
  sharedVars: Record<string, number>,
  sync: SyncContext = EMPTY_SYNC,
): SimEvent[] {
  const id = thread.def.id;
  const index = thread.pc;
  const instruction = thread.def.program[index];

  if (instruction === undefined) {
    throw new Error(
      `Thread "${id}" tidak punya instruksi pada indeks ${index}`,
    );
  }

  const events: SimEvent[] = [
    {
      type: 'INSTRUCTION',
      tick,
      thread: id,
      index,
      op: instruction.op,
    },
  ];

  const push = (event: SimEvent | null): void => {
    if (event) events.push(event);
  };

  let completed = true;

  switch (instruction.op) {
    case 'WORK': {
      if (thread.workRemaining === 0) {
        thread.workRemaining = instruction.ticks;
      }

      thread.workRemaining -= 1;
      completed = thread.workRemaining === 0;
      break;
    }

    case 'READ': {
      // Sengaja terpisah dari WRITE: jeda di antaranya yang memunculkan race condition.
      thread.registers[instruction.into] = readNumber(
        sharedVars,
        instruction.from,
        id,
        'variabel bersama',
      );
      break;
    }

    case 'COMPUTE': {
      const current = readNumber(
        thread.registers,
        instruction.reg,
        id,
        'register',
      );

      thread.registers[instruction.reg] = current + instruction.add;
      break;
    }

    case 'WRITE': {
      const oldValue = readNumber(
        sharedVars,
        instruction.to,
        id,
        'variabel bersama',
      );

      const newValue = readNumber(
        thread.registers,
        instruction.reg,
        id,
        'register',
      );

      sharedVars[instruction.to] = newValue;

      events.push({
        type: 'VAR_WRITTEN',
        tick,
        thread: id,
        name: instruction.to,
        oldValue,
        newValue,
      });

      break;
    }

    case 'LOCK': {
      const mutex = requireMutex(sync, instruction.mutex, id);

      if (mutex.lock(id)) {
        events.push({
          type: 'LOCK_ACQUIRED',
          tick,
          thread: id,
          mutex: instruction.mutex,
        });
      } else {
        completed = false;

        const on = `mutex:${instruction.mutex}`;
        thread.waitingOn = on;

        events.push({
          type: 'BLOCKED',
          tick,
          thread: id,
          on,
        });

        push(transition(thread, 'WAITING', tick));
      }

      break;
    }

    case 'UNLOCK': {
      const mutex = requireMutex(sync, instruction.mutex, id);

      events.push(
        ...releaseMutex(mutex, instruction.mutex, thread, tick, sync),
      );

      break;
    }

    case 'SEM_WAIT': {
      const sem = requireSemaphore(sync, instruction.sem, id);

      if (!sem.wait(id)) {
        completed = false;

        const on = `sem:${instruction.sem}`;
        thread.waitingOn = on;

        events.push({
          type: 'BLOCKED',
          tick,
          thread: id,
          on,
        });

        push(transition(thread, 'WAITING', tick));
      }

      break;
    }

    case 'SEM_SIGNAL': {
      const sem = requireSemaphore(sync, instruction.sem, id);
      const next = sem.signal();

      if (next !== null) {
        const woken = sync.getThread(next);

        push(transition(woken, 'READY', tick));
        push(advancePc(woken, tick));
      }

      break;
    }

    case 'COND_WAIT': {
      const cond = requireCondition(sync, instruction.cond, id);
      const mutex = requireMutex(sync, instruction.mutex, id);

      // 1. Lepas mutex (melempar error bila thread bukan pemiliknya).
      events.push(
        ...releaseMutex(mutex, instruction.mutex, thread, tick, sync),
      );

      // 2. Tidur di condition sampai disinyal.
      cond.wait(id, instruction.mutex);
      completed = false;

      const on = `cond:${instruction.cond}`;
      thread.waitingOn = on;

      events.push({
        type: 'BLOCKED',
        tick,
        thread: id,
        on,
      });

      push(transition(thread, 'WAITING', tick));

      break;
    }

    case 'COND_SIGNAL': {
      const cond = requireCondition(sync, instruction.cond, id);
      const waiter = cond.signal(); // null = tidak ada yang menunggu, sinyal hilang

      if (waiter !== null) {
        events.push(...wakeWaiter(waiter, tick, sync));
      }

      break;
    }

    case 'COND_BROADCAST': {
      const cond = requireCondition(sync, instruction.cond, id);

      for (const waiter of cond.broadcast()) {
        events.push(...wakeWaiter(waiter, tick, sync));
      }

      break;
    }

    case 'ENTER_AREA': {
      sync.enterArea?.(id, instruction.area);

      events.push({
        type: 'AREA_ENTERED',
        tick,
        thread: id,
        area: instruction.area,
      });

      break;
    }

    case 'LEAVE_AREA': {
      sync.leaveArea?.(id, instruction.area);

      events.push({
        type: 'AREA_LEFT',
        tick,
        thread: id,
        area: instruction.area,
      });

      break;
    }

    case 'PUT_ITEM': {
      const buffer = requireBuffer(sync, instruction.buffer, id);

      if (buffer.put()) {
        events.push({
          type: 'BUFFER_PUT',
          tick,
          thread: id,
          buffer: instruction.buffer,
          items: buffer.count,
        });
      } else {
        events.push({
          type: 'BUFFER_OVERFLOW',
          tick,
          thread: id,
          buffer: instruction.buffer,
          capacity: buffer.capacity,
        });
      }

      break;
    }

    case 'TAKE_ITEM': {
      const buffer = requireBuffer(sync, instruction.buffer, id);

      if (buffer.take()) {
        events.push({
          type: 'BUFFER_TAKEN',
          tick,
          thread: id,
          buffer: instruction.buffer,
          items: buffer.count,
        });
      } else {
        events.push({
          type: 'BUFFER_UNDERFLOW',
          tick,
          thread: id,
          buffer: instruction.buffer,
        });
      }

      break;
    }

    case 'RW_ACQUIRE_READ':
    case 'RW_ACQUIRE_WRITE': {
      const lock = requireRwLock(sync, instruction.lock, id);
      const isRead = instruction.op === 'RW_ACQUIRE_READ';
      const granted = isRead ? lock.acquireRead(id) : lock.acquireWrite(id);

      if (granted) {
        events.push({
          type: 'RW_ACQUIRED',
          tick,
          thread: id,
          lock: instruction.lock,
          mode: isRead ? 'read' : 'write',
        });
      } else {
        completed = false;

        const on = `rw:${instruction.lock}`;
        thread.waitingOn = on;

        events.push({ type: 'BLOCKED', tick, thread: id, on });

        push(transition(thread, 'WAITING', tick));
      }

      break;
    }

    case 'RW_RELEASE_READ':
    case 'RW_RELEASE_WRITE': {
      const lock = requireRwLock(sync, instruction.lock, id);
      const isRead = instruction.op === 'RW_RELEASE_READ';
      const granted = isRead ? lock.releaseRead(id) : lock.releaseWrite(id);

      events.push({
        type: 'RW_RELEASED',
        tick,
        thread: id,
        lock: instruction.lock,
        mode: isRead ? 'read' : 'write',
      });
      events.push(
        ...wakeRwGranted(lock, instruction.lock, granted, tick, sync),
      );

      break;
    }

    case 'BOARD_READ': {
      const board = requireBoard(sync, instruction.board, id);

      events.push({
        type: 'BOARD_READ',
        tick,
        thread: id,
        board: instruction.board,
        consistent: board.read(),
      });

      break;
    }

    case 'BOARD_WRITE_START': {
      const board = requireBoard(sync, instruction.board, id);

      events.push({
        type: 'BOARD_WRITE',
        tick,
        thread: id,
        board: instruction.board,
        phase: 'start',
        conflict: board.startWrite(),
      });

      break;
    }

    case 'BOARD_WRITE_END': {
      const board = requireBoard(sync, instruction.board, id);

      board.endWrite();

      events.push({
        type: 'BOARD_WRITE',
        tick,
        thread: id,
        board: instruction.board,
        phase: 'end',
        conflict: false,
      });

      break;
    }

    case 'SPIN_UNTIL': {
      // Busy waiting: thread tetap READY/RUNNING dan memakai core tiap tick
      // selama kondisi belum terpenuhi. Beda dengan WAITING (tidur, tanpa core).
      // Tick pengecekan yang gagal dihitung sebagai busyWaitTicks; tick
      // pengecekan yang berhasil sudah dianggap pekerjaan biasa.
      const current = readNumber(
        sharedVars,
        instruction.variable,
        id,
        'variabel bersama',
      );

      if (current < instruction.atLeast) {
        completed = false;
        thread.busyWaitTicks += 1;
      }

      break;
    }

    default: {
      // Semua instruksi sudah ditangani; bila union Instruction bertambah,
      // TypeScript akan menandai baris ini sebagai error.
      const unhandled: never = instruction;

      throw new Error(
        `Instruksi belum didukung engine: ${JSON.stringify(unhandled)}`,
      );
    }
  }

  if (completed) push(advancePc(thread, tick));

  return events;
}

/**
 * Sisa "kesabaran" thread sebagai pecahan 0..1 (1 = baru mulai menunggu atau
 * tidak menunggu, 0 = sudah mencapai ambang starvation). Dipakai bar kesabaran
 * writer di Level 2-2. Seperti `batteryLevel`: fungsi murni, tampilan hanya
 * menggambar angka ini dan tidak menghitung apa pun sendiri.
 */
export function patienceLevel(
  waitStreak: number,
  thresholdTicks: number,
): number {
  if (!(thresholdTicks > 0)) {
    throw new Error(
      `thresholdTicks harus lebih dari 0, diterima: ${thresholdTicks}`,
    );
  }
  const used = Math.max(0, waitStreak) / thresholdTicks;
  return Math.min(1, Math.max(0, 1 - used));
}

/**
 * Thread yang pernah menunggu terus-menerus minimal `starvationTicks` tick
 * (diukur dari rentetan terpanjang, jadi tetap tercatat walau sudah dibangunkan).
 * Daftar kosong bila deteksi tidak dipakai (ambang tidak diatur) atau snapshot
 * belum memuat data rentetan. Urutan mengikuti urutan thread di snapshot.
 */
export function starvedThreads(snapshot: Readonly<Snapshot>): ThreadId[] {
  const threshold = snapshot.starvationTicks;
  const peaks = snapshot.peakWaitStreaks;
  if (threshold === null || threshold === undefined || peaks === undefined) {
    return [];
  }
  return snapshot.threads
    .filter((thread) => (peaks[thread.id] ?? 0) >= threshold)
    .map((thread) => thread.id);
}
import type { Prng } from './prng';
import type { SchedulerConfig, ThreadId } from './types';

/**
 * Si Penjadwal: tiap tick memilih thread mana (maksimal `cores`) yang boleh
 * mengeksekusi satu instruksi. Thread lain "dibekukan" pada tick itu.
 */
export interface Scheduler {
  /**
   * @param runnable id thread yang boleh jalan (READY atau RUNNING), dalam
   *                 urutan definisi thread (urutan stabil demi determinisme).
   * @returns id thread terpilih, tanpa duplikat, panjang <= cores. Selalu
   *          minimal satu bila `runnable` tidak kosong.
   */
  pick(runnable: readonly ThreadId[]): ThreadId[];
}

/** Memilih acak (dari PRNG berbasis seed) di antara thread yang boleh jalan. */
function createRandomScheduler(cores: number, prng: Prng): Scheduler {
  return {
    pick(runnable) {
      const pool = [...runnable];
      const picked: ThreadId[] = [];
      while (picked.length < cores && pool.length > 0) {
        const chosen = prng.pick(pool);
        picked.push(chosen);
        pool.splice(pool.indexOf(chosen), 1);
      }
      return picked;
    },
  };
}

/**
 * Mengikuti urutan `script` (satu entri = satu jatah eksekusi). Dipakai di level
 * tutorial agar race condition PASTI muncul.
 *
 * Aturan:
 * - Entri untuk thread yang tidak bisa jalan (WAITING/TERMINATED) dilewati,
 *   supaya script tidak membuat simulasi macet.
 * - Dengan cores > 1, satu tick mengambil entri berurutan; bila ketemu thread
 *   yang sudah terpilih pada tick itu, tick berakhir dan entri tersebut
 *   dipakai di tick berikutnya.
 * - Setelah script habis, sisa jatah diisi thread yang boleh jalan menurut
 *   urutan definisi, sehingga simulasi tetap bisa selesai.
 */
function createScriptedScheduler(
  cores: number,
  script: readonly ThreadId[],
): Scheduler {
  let cursor = 0;
  return {
    pick(runnable) {
      const runnableSet = new Set(runnable);
      const picked: ThreadId[] = [];

      while (picked.length < cores && cursor < script.length) {
        const id = script[cursor] as ThreadId;
        if (picked.includes(id)) break;
        cursor += 1;
        if (runnableSet.has(id)) picked.push(id);
      }

      // Script habis: isi sisa jatah secara deterministik.
      if (cursor >= script.length) {
        for (const id of runnable) {
          if (picked.length >= cores) break;
          if (!picked.includes(id)) picked.push(id);
        }
      }
      return picked;
    },
  };
}

/**
 * Membuat scheduler sesuai konfigurasi. Panggil ulang tiap load()/reset()
 * karena scheduler menyimpan keadaan (posisi script, PRNG).
 */
export function createScheduler(
  config: SchedulerConfig,
  prng: Prng,
): Scheduler {
  switch (config.kind) {
    case 'random':
      return createRandomScheduler(config.cores, prng);
    case 'scripted':
      return createScriptedScheduler(config.cores, config.script);
  }
}
import type { RwLockSnapshot, RwMode, RwPolicy } from './primitives/rwlock';
export type ThreadId = string;

/**
 * Status thread.
 * NEW -> READY -> RUNNING -> (WAITING <-> READY) -> TERMINATED
 * WAITING = diblokir (menunggu mutex/semaphore/condition) dan TIDAK dipilih scheduler.
 */
export type ThreadState =
  'NEW' | 'READY' | 'RUNNING' | 'WAITING' | 'TERMINATED';

/**
 * Satu instruksi dalam program sebuah thread. Satu tick = satu instruksi.
 */
export type Instruction =
  | { op: 'WORK'; ticks: number } // pekerjaan lokal, tanpa data bersama
  | { op: 'READ'; from: string; into: string } // shared var -> register
  | { op: 'COMPUTE'; reg: string; add: number } // register += add
  | { op: 'WRITE'; reg: string; to: string } // register -> shared var
  | { op: 'LOCK'; mutex: string }
  | { op: 'UNLOCK'; mutex: string }
  | { op: 'SEM_WAIT'; sem: string }
  | { op: 'SEM_SIGNAL'; sem: string }
  | { op: 'COND_WAIT'; cond: string; mutex: string }
  | { op: 'COND_SIGNAL'; cond: string }
  | { op: 'COND_BROADCAST'; cond: string }
  | { op: 'ENTER_AREA'; area: string }
  | { op: 'LEAVE_AREA'; area: string }

  // Ban berjalan (T-048). Tidak memblokir; pembatasnya semaphore empty/full.
  | { op: 'PUT_ITEM'; buffer: string } // taruh satu barang; ban penuh = overflow
  | { op: 'TAKE_ITEM'; buffer: string } // ambil satu barang; ban kosong = underflow

  // Readers-Writers (T-051, Level 2-2). Acquire bisa memblokir (WAITING).
  | { op: 'RW_ACQUIRE_READ'; lock: string }
  | { op: 'RW_RELEASE_READ'; lock: string }
  | { op: 'RW_ACQUIRE_WRITE'; lock: string }
  | { op: 'RW_RELEASE_WRITE'; lock: string }

  // Papan pengumuman: penulisan dua langkah; bacaan di antaranya setengah jadi.
  | { op: 'BOARD_READ'; board: string }
  | { op: 'BOARD_WRITE_START'; board: string }
  | { op: 'BOARD_WRITE_END'; board: string }
  // Busy waiting (T-045): tiap tick mengecek variabel bersama; selama nilainya
  // masih di bawah atLeast, thread TIDAK tidur dan terus memakai core.
  | { op: 'SPIN_UNTIL'; variable: string; atLeast: number };

/** Nama operasi saja, mis. 'LOCK'. Dipakai di event INSTRUCTION. */
export type Op = Instruction['op'];

/** Definisi sebuah thread: program + jumlah pengulangan. */
export interface ThreadDef {
  id: ThreadId;

  /** Nama tampilan di layar, mis. 'Robot A'. */
  label: string;

  program: Instruction[];

  /** Berapa kali program dijalankan berulang. Default 1. */
  repeat?: number;
}

/**
 * Pengaturan scheduler.
 * - random   : tiap tick memilih acak (dari PRNG berbasis seed) di antara thread READY.
 * - scripted : urutan eksekusi diberikan eksplisit (agar race condition PASTI muncul
 *              di level tutorial). Detail perilaku saat script habis diatur di T-014.
 * cores = jumlah thread yang boleh mengeksekusi instruksi dalam satu tick.
 */
export type SchedulerConfig =
  | { kind: 'random'; cores: number }
  | { kind: 'scripted'; cores: number; script: ThreadId[] };

/**
 * Semua yang dibutuhkan engine untuk menjalankan satu simulasi.
 *
 * Catatan: engine TIDAK mengimpor LevelDefinition dari levels/ (supaya tidak
 * ada ketergantungan melingkar). Controller yang menerjemahkan
 * LevelDefinition + alat pilihan pemain menjadi EngineConfig ini.
 */
export interface EngineConfig {
  threads: ThreadDef[];

  /** Nilai awal variabel bersama, mis. { stok: 0 }. */
  sharedVars: Record<string, number>;

  /** Daftar nama mutex. */
  mutexes: string[];

  /** Nama semaphore -> nilai awal. */
  semaphores: Record<string, number>;

  /** Daftar nama condition variable. */
  conditions: string[];

  scheduler: SchedulerConfig;

  /** Kapasitas area berkapasitas, mis. { aula: 3 }. Bila tidak dispesifikasikan, default {}. */
  areas?: Record<string, number>;

  /** Kapasitas ban berjalan, mis. { ban: 3 }. Bila tidak ada, default {}. */
  buffers?: Record<string, number>;

  /** Kunci pembaca/penulis: nama -> kebijakan (Level 2-2). Bila tidak ada, default {}. */
  rwLocks?: Record<string, RwPolicy>;

  /** Nama papan pengumuman (Level 2-2). Bila tidak ada, default []. */
  boards?: string[];

  /**
   * Ambang starvation (T-053): thread yang menunggu TERUS-MENERUS minimal
   * sebanyak ini tick dianggap kelaparan. Bila tidak ada, deteksi mati
   * (waitStreaks tetap dihitung). Bilangan bulat >= 1.
   */
  starvationTicks?: number;

  /** Batas pengaman: simulasi dihentikan paksa setelah tick ini. */
  maxTicks: number;
}

export interface ThreadSnapshot {
  id: ThreadId;
  label: string;
  state: ThreadState;

  /** Indeks instruksi berikutnya yang akan dieksekusi. */
  pc: number;

  /** Pengulangan program ke berapa (mulai 0). */
  iteration: number;

  registers: Record<string, number>;

  /** Nama resource yang sedang ditunggu (mis. 'mutex:papan'), atau null. */
  waitingOn: string | null;

  /** Sisa tick untuk instruksi WORK yang sedang berjalan. */
  workRemaining: number;

  /** Total tick yang sudah dihabiskan di state WAITING. */
  waitTicks: number;

  busyWaitTicks: number;
}

export interface MutexSnapshot {
  id: string;
  owner: ThreadId | null;
  queue: ThreadId[];
}

export interface SemaphoreSnapshot {
  id: string;
  value: number;
  queue: ThreadId[];
}

export interface ConditionSnapshot {
  id: string;
  queue: ThreadId[];
}

export interface BoardSnapshot {
  id: string;

  /** Jumlah penulis yang sedang menulis (0 = papan utuh). */
  activeWriters: number;

  /** Berapa kali penulisan selesai. */
  version: number;

  reads: number;

  /** Bacaan saat papan sedang ditulis (data setengah jadi). */
  inconsistentReads: number;

  /** Penulisan yang dimulai saat penulis lain masih menulis. */
  writeConflicts: number;
}

export interface BufferSnapshot {
  id: string;
  capacity: number;

  /** Barang di ban saat ini. */
  items: number;

  /** Berapa kali PUT_ITEM pada ban penuh (barang jatuh). */
  overflows: number;

  /** Berapa kali TAKE_ITEM pada ban kosong (tangan meraih kosong). */
  underflows: number;

  /** Barang yang berhasil ditaruh dan diambil. */
  totalPut: number;
  totalTaken: number;
}

/**
 * Keadaan penuh simulasi. Bagi pemanggil (controller, Phaser, React) snapshot
 * bersifat BACA-SAJA: jangan dimodifikasi.
 *
 * Metrik (throughput, lostUpdates, dst.) dihitung terpisah oleh metrics.ts (computeMetrics).
 */
export interface Snapshot {
  tick: number;
  seed: number;
  busyWaitTicks: number;
  finished: boolean;

  threads: ThreadSnapshot[];
  sharedVars: Record<string, number>;
  mutexes: MutexSnapshot[];
  semaphores: SemaphoreSnapshot[];
  conditions: ConditionSnapshot[];

  /** Jumlah thread yang saat ini berada di dalam tiap area, mis. { aula: 2 }. */
  areaOccupancy: Record<string, number>;

  /** Total tick dengan kejadian overload kapasitas area. */
  overloadTicks: number;

  /** Rata-rata utilisasi area (0..1). */
  areaUtilization: number;

  /** Jumlah core yang dipakai scheduler (dari konfigurasi level). */
  cores: number;

  /** Total eksekusi instruksi sejak tick 0 (satu per thread terpilih per tick). */
  executedInstructions: number;

  /** Keadaan tiap ban berjalan (kosong bila level tanpa ban). */
  buffers: BufferSnapshot[];

  /** Jumlah overflow dan underflow dari semua ban. */
  overflowCount: number;
  underflowCount: number;

  /** Papan pengumuman (Level 2-2); kosong bila level tanpa papan. */
  boards: BoardSnapshot[];

  /** Kunci pembaca/penulis (Level 2-2); kosong bila level tanpa kunci itu. */
  rwLocks: RwLockSnapshot[];

  /** Jumlah bacaan papan saat sedang ditulis, dari semua papan. */
  inconsistentReadCount: number;

  /** Jumlah penulisan papan yang bertabrakan, dari semua papan. */
  writeConflictCount: number;

  // Starvation (T-053). Ketiganya opsional supaya fixture snapshot buatan
  // tangan di tes lama tetap valid; Engine selalu mengisinya.

  /**
   * Tick menunggu BERTURUT-TURUT saat ini per thread (0 bila tidak sedang
   * WAITING). Beda dengan ThreadSnapshot.waitTicks yang kumulatif: yang ini
   * kembali ke 0 begitu thread dibangunkan.
   */
  waitStreaks?: Record<ThreadId, number>;

  /** Rentetan menunggu terpanjang yang pernah dialami tiap thread. */
  peakWaitStreaks?: Record<ThreadId, number>;

  /** Ambang starvation dari konfigurasi; null bila deteksi tidak dipakai. */
  starvationTicks?: number | null;
}

/** Event dipancarkan engine; Phaser menganimasikannya, React menampilkannya di log. */
export type SimEvent =
  | {
      type: 'THREAD_STATE';
      tick: number;
      thread: ThreadId;
      from: ThreadState;
      to: ThreadState;
    }
  | {
      type: 'INSTRUCTION';
      tick: number;
      thread: ThreadId;
      index: number;
      op: Op;
    }
  | {
      type: 'VAR_WRITTEN';
      tick: number;
      thread: ThreadId;
      name: string;
      oldValue: number;
      newValue: number;
    }
  | {
      type: 'LOCK_ACQUIRED';
      tick: number;
      thread: ThreadId;
      mutex: string;
    }
  | {
      type: 'LOCK_RELEASED';
      tick: number;
      thread: ThreadId;
      mutex: string;
    }

  /** Thread berhenti menunggu resource on (mis. 'mutex:papan'). */
  | {
      type: 'BLOCKED';
      tick: number;
      thread: ThreadId;
      on: string;
    }
  | {
      type: 'DEADLOCK_DETECTED';
      tick: number;
      cycle: ThreadId[];
    }
  | {
      type: 'AREA_ENTERED';

import type { ThreadId } from '../types';

/** Kebijakan pembagian akses (PRD Level 2-2). */
export type RwPolicy = 'reader' | 'writer' | 'fair';
export type RwMode = 'read' | 'write';

export interface RwWaiter {
  thread: ThreadId;
  mode: RwMode;
}

export interface RwLockSnapshot {
  id: string;
  policy: RwPolicy;
  readers: ThreadId[];
  writer: ThreadId | null;
  queue: RwWaiter[];
}

/**
 * Kunci banyak-pembaca/satu-penulis (Readers-Writers, Level 2-2).
 * Banyak pembaca boleh bersamaan; penulis harus sendirian. Kebijakan menentukan
 * siapa didahulukan saat ada yang menunggu:
 * - reader: pembaca baru boleh masuk selama tidak ada penulis AKTIF, sehingga
 *   penulis bisa kelaparan (starvation);
 * - writer: bila ada penulis menunggu, pembaca baru antre di belakangnya;
 * - fair: satu antrean FIFO; pembaca berurutan di depan masuk bersamaan.
 * Seperti Mutex, thread yang dibangunkan langsung menjadi pemegang (handoff),
 * jadi antrean tidak pernah dilangkahi secara diam-diam.
 */
export class RwLock {
  private readers = new Set<ThreadId>();
  private writer: ThreadId | null = null;
  private queue: RwWaiter[] = [];

  constructor(
    readonly id: string,
    readonly policy: RwPolicy,
  ) {}

  /** @returns true bila langsung masuk; false bila harus antre (diblokir). */
  acquireRead(thread: ThreadId): boolean {
    this.assertIdle(thread);
    if (this.writer === null && this.canReaderJoin()) {
      this.readers.add(thread);
      return true;
    }
    this.queue.push({ thread, mode: 'read' });
    return false;
  }

  /** @returns true bila langsung masuk; false bila harus antre (diblokir). */
  acquireWrite(thread: ThreadId): boolean {
    this.assertIdle(thread);
    if (
      this.writer === null &&
      this.readers.size === 0 &&
      this.queue.length === 0
    ) {
      this.writer = thread;
      return true;
    }
    this.queue.push({ thread, mode: 'write' });
    return false;
  }

  /** @returns thread yang dibangunkan (sudah menjadi pemegang), urut antrean. */
  releaseRead(thread: ThreadId): ThreadId[] {
    if (!this.readers.delete(thread)) {
      throw new Error(
        `Thread "${thread}" bukan pembaca aktif di "${this.id}" (RW_RELEASE_READ salah)`,
      );
    }
    return this.dispatch();
  }

  /** @returns thread yang dibangunkan (sudah menjadi pemegang), urut antrean. */
  releaseWrite(thread: ThreadId): ThreadId[] {
    if (this.writer !== thread) {
      throw new Error(
        `Thread "${thread}" bukan penulis aktif di "${this.id}" (RW_RELEASE_WRITE salah)`,
      );
    }
    this.writer = null;
    return this.dispatch();
  }

  snapshot(): RwLockSnapshot {
    return {
      id: this.id,
      policy: this.policy,
      readers: [...this.readers],
      writer: this.writer,
      queue: this.queue.map((waiter) => ({ ...waiter })),
    };
  }

  private assertIdle(thread: ThreadId): void {
    if (
      this.readers.has(thread) ||
      this.writer === thread ||
      this.queue.some((waiter) => waiter.thread === thread)
    ) {
      throw new Error(
        `Thread "${thread}" sudah memegang atau mengantre di "${this.id}"`,
      );
    }
  }

  /** Bolehkah pembaca baru masuk (dengan asumsi tidak ada penulis aktif)? */
  private canReaderJoin(): boolean {
    if (this.policy === 'reader') return true;
    if (this.policy === 'writer') {
      return !this.queue.some((waiter) => waiter.mode === 'write');
    }
    return this.queue.length === 0;
  }

  /** Menjadikan penulis terdepan di antrean sebagai pemegang bila tidak ada pembaca. */
  private admitFirstWriter(granted: ThreadId[]): void {
    if (this.readers.size > 0) return;
    const index = this.queue.findIndex((waiter) => waiter.mode === 'write');
    const waiter = this.queue[index];
    if (waiter === undefined) return;
    this.queue.splice(index, 1);
    this.writer = waiter.thread;
    granted.push(waiter.thread);
  }

  /** Membagikan kunci kepada yang menunggu menurut kebijakan. */
  private dispatch(): ThreadId[] {
    const granted: ThreadId[] = [];
    if (this.writer !== null) return granted;

    const admitReaders = (waiters: RwWaiter[]): void => {
      for (const waiter of waiters) {
        this.readers.add(waiter.thread);
        granted.push(waiter.thread);
      }
    };

    if (this.policy === 'reader') {
      const waitingReaders = this.queue.filter((w) => w.mode === 'read');
      if (waitingReaders.length > 0) {
        admitReaders(waitingReaders);
        this.queue = this.queue.filter((w) => w.mode === 'write');
      } else {
        this.admitFirstWriter(granted);
      }
      return granted;
    }

    if (this.policy === 'writer') {
      if (this.queue.some((w) => w.mode === 'write')) {
        this.admitFirstWriter(granted);
      } else {
        admitReaders(this.queue);
        this.queue = [];
      }
      return granted;
    }

    // fair: layani dari depan antrean; berhenti di penulis yang belum boleh masuk.
    for (;;) {
      const head = this.queue[0];
      if (head === undefined) break;
      if (head.mode === 'read') {
        this.queue.shift();
        admitReaders([head]);
        continue;
      }
      this.admitFirstWriter(granted);
      break;
    }
    return granted;
  }
}

  threads: ThreadDef[],

  const engine = new Engine();

  return engine.getSnapshot();

const THREADS: ThreadDef[] = [

/** Menjalankan sampai selesai; mengembalikan riwayat rentetan W tiap tick. */
function run(cfg: EngineConfig, seed = 1) {
  const engine = new Engine();
  engine.load(cfg, seed);
  const history: { state: string; streak: number }[] = [];
  while (!engine.isFinished) {
    engine.step();
    const snap = engine.getSnapshot();
    history.push({
      state: snap.threads.find((t) => t.id === 'W')?.state ?? '',
      streak: snap.waitStreaks?.['W'] ?? -1,
    });
  }
  return { engine, history, snapshot: engine.getSnapshot() };
}

describe('rentetan menunggu (waitStreaks)', () => {
  it('bertambah tepat 1 per tick selama WAITING', () => {
    const { history } = run(config());
    const waiting = history
      .map((h, i) => ({ ...h, i }))
      .filter((h) => h.state === 'WAITING');
    expect(waiting.length).toBeGreaterThanOrEqual(2);
    // Snapshot diambil sesudah tick; W yang baru diblokir pada tick itu belum
    // dihitung (penghitungan terjadi di awal tick berikutnya), jadi cukup
    // memeriksa selisih antar tick WAITING yang berurutan.
    for (let k = 1; k < waiting.length; k += 1) {
      const prev = waiting[k - 1];
      const cur = waiting[k];
      if (prev === undefined || cur === undefined) continue;
      if (cur.i === prev.i + 1) expect(cur.streak).toBe(prev.streak + 1);
    }
  });

  it('kembali ke 0 setelah thread dibangunkan', () => {
    const { snapshot } = run(config());
    expect(snapshot.threads.every((t) => t.state === 'TERMINATED')).toBe(true);
    expect(snapshot.waitStreaks?.['W']).toBe(0);
    expect(snapshot.waitStreaks?.['R1']).toBe(0);
  });

  it('peakWaitStreaks menyimpan rentetan terpanjang dan tidak melebihi waitTicks', () => {
    const { history, snapshot } = run(config());
    const observedMax = Math.max(...history.map((h) => h.streak));
    const peak = snapshot.peakWaitStreaks?.['W'] ?? -1;
    const waitTicks = snapshot.threads.find((t) => t.id === 'W')?.waitTicks;
    expect(peak).toBeGreaterThanOrEqual(observedMax);
    expect(peak).toBeGreaterThan(0);
    expect(peak).toBeLessThanOrEqual(waitTicks ?? -1);
    // W hanya satu kali menunggu, jadi rentetan terpanjang = total menunggu.
    expect(peak).toBe(waitTicks);
  });

  it('thread yang tidak pernah menunggu punya puncak 0', () => {
    const { snapshot } = run(config());
    expect(snapshot.peakWaitStreaks?.['R1']).toBe(0);
  });

  it('seed sama menghasilkan rentetan yang sama persis', () => {
    const a = run(config(), 7).snapshot;
    const b = run(config(), 7).snapshot;
    expect(a.waitStreaks).toEqual(b.waitStreaks);
    expect(a.peakWaitStreaks).toEqual(b.peakWaitStreaks);
  });

  it('load ulang mengosongkan rentetan', () => {
    const { engine } = run(config());
    engine.reset();
    const snap = engine.getSnapshot();
    expect(snap.waitStreaks).toEqual({ R1: 0, W: 0 });
    expect(snap.peakWaitStreaks).toEqual({ R1: 0, W: 0 });
  });

  it('snapshot adalah salinan: mengubahnya tidak memengaruhi engine', () => {
    const { engine } = run(config());
    const snap = engine.getSnapshot();
    if (snap.peakWaitStreaks !== undefined) snap.peakWaitStreaks['W'] = 999;
    expect(engine.getSnapshot().peakWaitStreaks?.['W']).not.toBe(999);
  });
});
